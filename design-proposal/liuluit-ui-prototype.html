// 清晰的依赖边界，比聪明的代码更重要
struct ContentStore {
  private let repo: ArticleRepository
  private let cache: Cache

  func load(_ id: Article.ID) async throws -> Article {
    if let cached = try cache.get(id) { return cached }
    let article = try await repo.fetch(id)
    try cache.set(article)
    return article
  }
}
